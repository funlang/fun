# Tests skipped on the Linux LoongArch64 qemu-user regression.
#
# These are emulation-host limits, not architecture failures:
#   - lib-asm / lib-jit / lib-tcc: no LoongArch backend yet.
#   - the lib-fun-pack/package/test/uuid shell tests invoke the target binary
#     directly instead of through FUNCMD, so they cannot run under qemu-user.
#   - run-ado-odbc.sh needs an ODBC driver; the Alpine sysroot only carries the
#     runtime libraries used by the Fun stdlib.
#   - lib-loadcheck / lib-func-check / run-benchmarks depend on the full host
#     toolchain and module set.
#
# lib-ajax-lnx-test is NOT skipped: it has a linux-loongarch64-qemu snapshot
# because Alpine's libcurl says "Could not connect" instead of "Couldn't
# connect".

# --- native code generation -------------------------------------------------
lib-asm-lnx.fun	x86-64 assembly backend
lib-asm-x64.fun	x86-64 assembly backend
fractal-asm-x64.fun	x86-64 assembly backend
lib-asm-a64.fun	aarch64 assembly backend, wrong cpu
lib-jit-test.fun	JIT backend emits x86-64
lib-tcc-test.fun	no TinyCC LoongArch backend

# --- libffi closures under qemu-user ---------------------------------------
# @toCallback(..., ptr: true) executes a libffi closure trampoline. qemu-user
# 7.2/8.2 can hang on the generated code instead of returning, so the test
# cannot complete in emulation. It passes on native LoongArch hardware.
raw-address-getapi.fun	libffi closure trampoline hangs under qemu-user

# The FFI helper library is built by Zig for loongarch64-musl. qemu-user
# aborts when a helper entry point uses the floating-point ABI (summ,
# callbacks), although plain libc calls and the capability probe work.
# Run these six on real hardware.
ffi-scalar.fun	qemu-user aborts in the Zig-built FFI helper
ffi-int64.fun	qemu-user aborts in the Zig-built FFI helper
ffi-float.fun	qemu-user aborts in the Zig-built FFI helper
ffi-float-cb.fun	qemu-user aborts in the Zig-built FFI helper
ffi-callbacks.fun	qemu-user aborts in the Zig-built FFI helper
ffi-var-ref.fun	qemu-user aborts in the Zig-built FFI helper

# --- shell project tests that execute the target binary directly -----------
lib-fun-pack-test.fun	invokes funcmd directly, not through FUNCMD
lib-fun-package-test.fun	invokes funcmd directly, not through FUNCMD
lib-fun-test-test.fun	invokes funcmd directly, not through FUNCMD
lib-fun-uuid-test.fun	invokes funcmd directly, not through FUNCMD
lib-cli-stdin-test.sh	invokes funcmd directly, not through FUNCMD
lib-io-stdio-test.sh	invokes funcmd directly, not through FUNCMD

# --- qemu-user /proc differences -------------------------------------------
# Under qemu-user /proc/<pid>/comm and /proc/<pid>/exe describe the host qemu
# process, not the guest funcmd, so these two process-name assertions cannot
# pass in emulation. They pass on native LoongArch hardware.
lib-proc-test.fun	/proc/<pid>/exe names the qemu host process
lib-os-lnx-test.fun	/proc/<pid>/comm names the qemu host process

# --- host toolchain / services not in the emulated sysroot -----------------
run-ado-odbc.sh	no ODBC driver in the emulated sysroot
lib-loadcheck.sh	module set/library availability
lib-func-check.fun	module set/library availability
run-benchmarks.sh	not a correctness test
